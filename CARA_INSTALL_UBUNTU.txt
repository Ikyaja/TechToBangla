Di Ubuntu Server, jalankan satu per satu:

# 1. Update system
sudo apt update && sudo apt upgrade -y

# 2. Install Python 3.10
sudo apt install -y python3.10 python3.10-venv python3-pip screen

# 3. Create directory
mkdir -p ~/bot_v6/{modules,config,logs}
cd ~/bot_v6

# 4. Create virtual environment
python3.10 -m venv venv
source venv/bin/activate

# 5. Install dependencies (pastikan venv aktif!)
pip install --upgrade pip
pip install python-telegram-bot==20.7
pip install python-binance==1.0.19
pip install python-dotenv==1.0.0
pip install aiohttp==3.9.1
pip install pandas==2.0.3 numpy==1.24.4

# 6. Copy bot files (dari komputer lokal):
# [Di komputer lokal, bukan di server!]
# scp /workspace/bot_v6_ultimate/main.py user@server-ip:~/bot_v6/
# scp /workspace/bot_v6_ultimate/modules/*.py user@server-ip:~/bot_v6/modules/
# scp /workspace/bot_v6_ultimate/config/.env user@server-ip:~/bot_v6/config/

# 7. Atau copy manual (di server):
nano main.py  # Paste content dari /workspace/bot_v6_ultimate/main.py
nano modules/technical_analysis_v6.py  # Paste content
nano modules/signal_generator_v6.py  # Paste content
nano modules/ui_formatter_v6.py  # Paste content
nano modules/binance_client.py  # Paste content
nano config/.env  # Paste content
touch __init__.py modules/__init__.py

# 8. Set permissions
chmod +x main.py
chmod 600 config/.env

# 9. Run bot!
python3 main.py

🎯 METODE 3: DENGAN SYSTEMD (24/7 Auto-Restart)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Setelah install dengan Metode 1 atau 2, setup systemd:

# 1. Create service file
sudo nano /etc/systemd/system/trading-bot-v6.service

# 2. Paste ini (GANTI 'ubuntu' dengan username Anda!):

[Unit]
Description=AI Future Signal Bot V6 Ultimate
After=network.target

[Service]
Type=simple
User=ubuntu
WorkingDirectory=/home/ubuntu/bot_v6
Environment="PATH=/home/ubuntu/bot_v6/venv/bin"
ExecStart=/home/ubuntu/bot_v6/venv/bin/python3 main.py
Restart=always
RestartSec=10
StandardOutput=append:/home/ubuntu/bot_v6/logs/output.log
StandardError=append:/home/ubuntu/bot_v6/logs/error.log

[Install]
WantedBy=multi-user.target

# 3. Save: Ctrl+O, Enter, Ctrl+X

# 4. Enable & start:
sudo systemctl daemon-reload
sudo systemctl enable trading-bot-v6
sudo systemctl start trading-bot-v6

# 5. Check status:
sudo systemctl status trading-bot-v6

# Should show:
# ● trading-bot-v6.service - AI Future Signal Bot V6 Ultimate
#    Loaded: loaded
#    Active: active (running)
